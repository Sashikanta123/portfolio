using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace Portfolio.Architecture;

[ApiController]
[Route("api/[controller]")]
public class TaskEngineController : ControllerBase
{
    private readonly AppDbContext _db;
    
    public TaskEngineController(AppDbContext db)
    {
        _db = db;
    }

    [HttpGet("performance-metrics")]
    public async Task<IActionResult> GetMetrics()
    {
        var metrics = await _db.Tasks
            .Where(t => t.IsActive)
            .GroupBy(t => t.Status)
            .Select(g => new {
                Status = g.Key,
                Count = g.Count()
            }).ToListAsync();

        return Ok(new {
            Engineer = "SHASHI KANTA SAHOO",
            Status = "Ready for Production",
            Data = metrics
        });
    }
}
